1. CLONAR O REPOSITÓRIO E SUBIR O PROJETO

git clone https://github.com/nomeDeUser/StreamFlow-Microsservices-madness.git
cd StreamFlow-Microsservices-madness

docker compose up --build
// constrói e inicia todos os containers. deixa esse terminal aberto.





2. ABRIR OUTRO CMD E PREPARAR O TOKEN, OS COMANDOS VÃO SER RODADOS NESSE TERMINAL, NÃO NO OUTRO

cd StreamFlow-Microsservices-madness

curl -s -X POST http://localhost:8080/auth/login -H "Content-Type: application/json" -d "{\"email\":\"admin@streamflow.com\",\"password\":\"admin123\"}"
// login e retorna JSON com token JWT

set TOKEN=COLA_SO_A_PARTE_DO_TOKEN_GERADO_ACIMA
// salvar token em uma variável temporária do CMD

curl -s http://localhost:8080/api/catalog -H "Authorization: Bearer %TOKEN%"
// testar se token funciona.. deve retornar a lista do catálogo.





3. PASTA PARA LOGS

mkdir evidencias





4. MEDIR LATÊNCIA DO CATÁLOGO

curl -w "\nTempo total: %{time_total}s\n" -o evidencias\catalog.json -s http://localhost:8080/api/catalog -H "Authorization: Bearer %TOKEN%"
// mede o tempo de uma rota simples e salva a resposta do catálogo





5. MEDIR LATÊNCIA DO PLAY

curl -w "\nTempo total: %{time_total}s\n" -o evidencias\play.json -s -X POST http://localhost:8080/api/streaming/play -H "Content-Type: application/json" -H "Authorization: Bearer %TOKEN%" -d "{\"movieId\":2}"
// mede o tempo do endpoint Play e salva a resposta

// Resultado obtido:
// Catálogo: 0.007135s
// Play:     0.259294s
// Diferença: ~252 ms
// Play foi aproximadamente 36 vezes mais lento.





6. TESTE COM NOTIFICATION-SERVICE PARADO

docker compose stop notification-service
// para apenas o serviço de notificações.

curl -s -X POST http://localhost:8080/api/streaming/play -H "Content-Type: application/json" -H "Authorization: Bearer %TOKEN%" -d "{\"movieId\":3}" > evidencias\play_sem_notification.json
// executa o Play sem o serviço de notificação e salva a resposta

type evidencias\play_sem_notification.json
// mstra a resposta salva.

docker compose logs streaming-service > evidencias\logs_streaming_sem_notification.txt
// salva os logs do streaming para registrar o erro de notificação.

docker compose start notification-service
// religa o serviço de notificações.

// Resultado observado:
// O Play retornou 201 mesmo sem notification-service.
// O log mostrou “Falha ao enviar notificação”.
// O tempo subiu para =~ 2,6s





7. TESTE COM BILLING-SERVICE PARADO

docker compose stop billing-service
// para apenas o serviço de faturamento

curl -s http://localhost:8080/api/analytics/report -H "Authorization: Bearer %TOKEN%" > evidencias\analytics_com_billing_parado.json
// executa o relatório do analytics com o billing parado

type evidencias\analytics_com_billing_parado.json
// mostra a resposta salva

docker compose logs analytics-service > evidencias\logs_analytics_com_billing_parado.txt
// salva os logs do analytics

docker compose logs billing-service > evidencias\logs_billing_parado.txt
// salva os logs do billing

docker compose start billing-service
// religa o billing-service.

// analytics continua respondendo, lê o banco compartilhado diretamente





8. HEALTH CHECK NORMAL
